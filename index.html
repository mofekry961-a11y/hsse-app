function doGet(e) { var action = e ? e.parameter.action : null; var result = {}; if (action === "getFaces") result = getAllFaceDescriptors(); else if (action === "getTopics") result = getTopicsList(e.parameter.mode || "Toolbox"); return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var result = {}; try { var postData = JSON.parse(e.postData.contents); if (postData.action === "checkWorker") result = getWorkerInfo(postData.workerId); else if (postData.action === "register") result = saveFaceDescriptor(postData.workerId, postData.descriptor); else if (postData.action === "record") result = recordAttendance(postData.sheetName, postData.workerId, postData.topicName); } catch (err) { result = { success: false, message: "خطأ: " + err.toString() }; } return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } function getWorkerInfo(workerId) { var data = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Data").getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0].toString().trim() === workerId.toString().trim()) { return { found: true, id: data[i][0], name: data[i][1], job: data[i][3] }; } } return { found: false }; } function saveFaceDescriptor(workerId, descriptorData) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Data"); var data = sheet.getDataRange().getValues(); var faceColIndex = data[0].indexOf("FaceData") === -1 ? data[0].length : data[0].indexOf("FaceData"); if (data[0].indexOf("FaceData") === -1) sheet.getRange(1, faceColIndex + 1).setValue("FaceData"); var descStr = typeof descriptorData === "object" ? JSON.stringify(descriptorData) : descriptorData; for (var i = 1; i < data.length; i++) { if (data[i][0].toString().trim() === workerId.toString().trim()) { sheet.getRange(i + 1, faceColIndex + 1).setValue(descStr); return { success: true, message: "تم حفظ البصمة بنجاح!" }; } } return { success: false, message: "لم يتم العثور على العامل" }; } function getAllFaceDescriptors() { var data = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Data").getDataRange().getValues(); var faceColIndex = data[0].indexOf("FaceData"); var list = []; if (faceColIndex === -1) return list; for (var i = 1; i < data.length; i++) { if (data[i][faceColIndex]) { try { list.push({ id: data[i][0], name: data[i][1], descriptor: JSON.parse(data[i][faceColIndex]) }); } catch(e) {} } } return list; } // الدالة السحرية الجديدة (النسخ التلقائي + منع التكرار) function recordAttendance(sheetName, workerId, topicName) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var dataSheet = ss.getSheetByName("Data"); var targetSheet = ss.getSheetByName(sheetName); if (!targetSheet) return { success: false, message: "الشيت غير موجود!" }; // 1. جلب بيانات العامل كاملة من شيت الداتا var dataValues = dataSheet.getDataRange().getValues(); var workerData = null; for (var i = 1; i < dataValues.length; i++) { if (dataValues[i][0].toString().trim() === workerId.toString().trim()) { workerData = dataValues[i]; // أخدنا الصف بتاعه بالكامل break; } } if (!workerData) return { success: false, message: "العامل غير مسجل في الداتا!" }; // 2. البحث عن عمود الموضوع في شيت الحضور var targetValues = targetSheet.getDataRange().getValues(); var headers = targetValues[0]; var topicCol = -1; for (var j = 0; j < headers.length; j++) { if (headers[j].toString().trim().toLowerCase() === topicName.toString().trim().toLowerCase()) { topicCol = j; break; } } if (topicCol === -1) return { success: false, message: "اسم المحاضرة غير موجود بالشيت!" }; var markValue = (sheetName === "Specific") ? "✓" : Utilities.formatDate(new Date(), "GMT+3", "yyyy-MM-dd HH:mm"); // 3. التأكد هل العامل ده نزل في الشيت ده قبل كده ولا لأ؟ var workerRow = -1; for (var k = 1; k < targetValues.length; k++) { if (targetValues[k][0].toString().trim() === workerId.toString().trim()) { workerRow = k + 1; break; } } if (workerRow !== -1) { // العامل موجود في الشيت.. هل حضر المحاضرة دي بالذات قبل كده؟ var existingValue = targetSheet.getRange(workerRow, topicCol + 1).getValue(); if (existingValue !== "") { return { success: false, message: "⚠️ العامل ده متسجل مسبقاً في نفس المحاضرة!" }; // منع التكرار } targetSheet.getRange(workerRow, topicCol + 1).setValue(markValue); // تسجيل الحضور } else { // العامل مش موجود خالص في الشيت.. انسخ بياناته من الداتا ونزله صف جديد var newRow = new Array(headers.length).fill(""); for (var m = 0; m <= 5; m++) { // نسخ أول 6 أعمدة (ID, الاسم, البطاقة, المهنة, التاريخ, الشركة) if (m < workerData.length) newRow[m] = workerData[m]; } newRow[topicCol] = markValue; targetSheet.appendRow(newRow); } return { success: true, workerName: workerData[1], topic: topicName }; } function getTopicsList(sheetName) { var headers = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName).getDataRange().getValues()[0]; return headers.slice(6).filter(function(h) { return h && h.toString().toLowerCase() !== "facedata"; }); }
