// دالة استقبال طلبات GET (فتح صفحة الأبلكيشن) function doGet() { return HtmlService.createTemplateFromFile('Index') .evaluate() .setTitle('نظام HSSE - بصمة الوجه') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // دالة استقبال طلبات POST (الحفظ والتسجيل من الموقع الخارجي) function doPost(e) { var data = JSON.parse(e.postData.contents); var action = data.action; var result = {}; if (action === "verify") { // البحث عن العامل وتأكيد الـ ID result = verifyWorker(data.workerId); } else if (action === "register") { // تسجيل الحضور (Toolbox أو Specific) result = registerWorker(data.sheetName, data.workerId, data.topicName); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 1. دالة التحقق من الـ ID في شيت Data function verifyWorker(workerId) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Data"); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0].toString().trim() === workerId.toString().trim()) { return { found: true, id: data[i][0], name: data[i][1], job: data[i][3], company: data[i][4] }; } } return { found: false, message: "الرقم التعريفي غير موجود" }; } // 2. دالة تسجيل الحضور (معدلة لمنع التكرار) function registerWorker(sheetName, workerId, topicName) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var targetSheet = ss.getSheetByName(sheetName); if (!targetSheet) return { success: false, message: "الشيت غير موجود" }; // أ. التأكد إن العامل موجود في شيت Data أولاً وسحب بياناته var workerInfo = verifyWorker(workerId); if (!workerInfo.found) return { success: false, message: "العامل غير مسجل في Data" }; // ب. البحث عن العامل في الشيت المستهدف (Toolbox/Specific) var targetData = targetSheet.getDataRange().getValues(); var targetRowIndex = -1; for (var j = 1; j < targetData.length; j++) { if (targetData[j][0].toString().trim() === workerId.toString().trim()) { targetRowIndex = j; break; } } // لو ملقيناهوش في شيت الـ Toolbox/Specific، نضيفه (نقل بياناته من Data) if (targetRowIndex === -1) { targetSheet.appendRow([ workerInfo.id, workerInfo.name, '', '', workerInfo.company, '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '' ]); targetData = targetSheet.getDataRange().getValues(); targetRowIndex = targetData.length - 1; } // ج. البحث عن عمود الموضوع var targetHeaders = targetData[0]; var topicColIndex = -1; for (var k = 0; k < targetHeaders.length; k++) { if (targetHeaders[k].toString().trim().toLowerCase() === topicName.toString().trim().toLowerCase()) { topicColIndex = k; break; } } if (topicColIndex === -1) { return { success: false, message: "الموضوع '" + topicName + "' غير موجود في عناوين الشيت!" }; } // د. منطق منع التكرار: نحاول نضيف التاريخ // نتحقق أولاً لو فيه تكرار بنفس التاريخ في أي خانة في الصف ده var todayDate = Utilities.formatDate(new Date(), "GMT+3", "yyyy-MM-dd"); var currentRow = targetData[targetRowIndex]; for (var c = 6; c < currentRow.length; c++) { // نبدأ من العمود 7 (المواضيع) var cellValue = currentRow[c].toString(); if (cellValue.indexOf(todayDate) !== -1) { return { success: false, message: "⛔ مكرر: العامل (" + workerInfo.name + ") مسجل بالفعل اليوم!" }; } } // هـ. حفظ التاريخ والوقت في أول خانة فاضية في الصف ده var timeNow = Utilities.formatDate(new Date(), "GMT+3", "yyyy-MM-dd HH:mm"); var inserted = false; for (var c = 6; c < currentRow.length; c++) { if (currentRow[c].toString().trim() === "") { targetSheet.getRange(targetRowIndex + 1, c + 1).setValue(timeNow); inserted = true; break; } } if (!inserted) { // لو مفيش خانة فاضية في المواضيع، نحط في العمود المختار (Overwrite or stop) targetSheet.getRange(targetRowIndex + 1, topicColIndex + 1).setValue(timeNow); } return { success: true, workerName: workerInfo.name, topic: topicName, time: timeNow }; }
